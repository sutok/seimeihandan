# 姓名判断アプリ - セキュリティ対策案

## 🚨 現在のセキュリティリスク

### 特定されたリスク
1. **無制限API アクセス**: 外部から誰でも直接リクエスト可能
2. **レート制限なし**: DoS攻撃・過負荷攻撃の脆弱性
3. **認証メカニズム不在**: 悪意のある利用者の制限不可
4. **監査ログ不備**: 不正利用の検知・追跡が困難

#### 2. シンプルレート制限
```python
import time
from collections import defaultdict

# IP別リクエスト記録
request_counts = defaultdict(list)

def check_rate_limit(client_ip, max_requests=10, window_minutes=1):
    now = time.time()
    window_start = now - (window_minutes * 60)
    
    # 古いリクエストを削除
    request_counts[client_ip] = [
        timestamp for timestamp in request_counts[client_ip] 
        if timestamp > window_start
    ]
    
    # 制限チェック
    if len(request_counts[client_ip]) >= max_requests:
        return False
    
    request_counts[client_ip].append(now)
    return True
```

#### 3. User-Agent検証

def validate_user_agent(request):
    user_agent = request.headers.get('User-Agent', '')
    
    # 明らかに怪しいUser-Agentをブロック
    blocked_patterns = ['bot', 'crawler', 'scraper', 'automated']
    suspicious_patterns = ['curl', 'wget', 'python-requests']
    
    for pattern in blocked_patterns:

def validate_request_payload(data):
    # ペイロードサイズ制限
    if len(json.dumps(data)) > 1024:  # 1KB制限
        return False, "Request too large"
    
    # 必須フィールド検証
    required_fields = ['lastName', 'firstName']
    for field in required_fields:
        if field not in data or not data[field]:
            return False, f"Missing required field: {field}"
    
    # 文字種・長さ制限
    for field in ['lastName', 'firstName']:
        value = data[field]
        if len(value) > 10:
            return False, f"{field} too long"
        
        # 日本語文字のみ許可

#### 3. 本格的レート制限（Redis使用）
```python
import redis
import json

redis_client = redis.Redis(host='localhost', port=6379, db=0)

def advanced_rate_limiting(client_ip, endpoint):
    # 複数時間窓でのレート制限
    limits = {
        'minute': {'requests': 10, 'window': 60},
        'hour': {'requests': 100, 'window': 3600},
        'day': {'requests': 1000, 'window': 86400}
    }
    
    for period, config in limits.items():
        key = f"rate_limit:{endpoint}:{client_ip}:{period}"
        current = redis_client.get(key)
        
        if current and int(current) >= config['requests']:
            return False, f"Rate limit exceeded: {period}"
        
        # カウンター更新
        pipe = redis_client.pipeline()
        pipe.incr(key)
        pipe.expire(key, config['window'])
        pipe.execute()
    
    return True, None
```
